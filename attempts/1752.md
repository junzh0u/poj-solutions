# 1752 — Advertisement

The special-judge gate was explicitly lifted for this run. The new suggested classification is `judge`: two independently structured, locally verified implementations receive System Error, matching the recent unrelated-user results. Existing front matter is preserved for the parent to stamp.

The input is one K,N pair followed by N paths. Normalize each pair because joggers can travel either direction. Both endpoints are included, coordinates range from -10000 to 10000, and each interval needs min(K,length) selected billboards. Output the minimum count and a strictly increasing list of distinct selected coordinates.

The primary algorithm sorts intervals by right endpoint and fills each deficit using its rightmost unused positions. An exchange argument proves optimality: an optimal extension containing earlier unused positions can move them right to the greedy choices without hurting already covered paths or any later interval. Scanning the 20001-coordinate domain for each of at most 1000 intervals costs O(N*20001) time and O(20001+N) space.

The independent control instead computes least feasible prefix-count potentials using longest-path difference constraints: S[r+1]>=S[l]+min(K,length), S[i+1]>=S[i], and S[i]>=S[i+1]-1. All potentials start at zero and queue relaxation finds the least feasible solution. Consecutive increments identify selected billboards. This is a canonical difference-constraints formulation, independently implemented rather than copied from the primary greedy algorithm. The discussion board mentions both approaches; its posted RE source at message 113739 was not used as a known-good control.

```cpp
// POJ 1752 - Advertisement
// Model: gpt-6-astra
// Process paths by increasing right endpoint, adding the rightmost unused
// billboards when necessary. An exchange to the right cannot harm a later
// path, so this greedy construction uses the minimum number of advertisements.
// Endpoints may be reversed or negative; paths shorter than K need all signs.
#include <cstdio>
#include <algorithm>
using namespace std;
struct Path { int l, r; };
bool byRight(const Path &a, const Path &b) {
    if (a.r != b.r) return a.r < b.r;
    return a.l < b.l;
}
Path paths[1000];
bool chosen[20001];
int main() {
    int k, n;
    if (scanf("%d%d", &k, &n) != 2) return 0;
    for (int i = 0; i < n; ++i) {
        int a, b;
        scanf("%d%d", &a, &b);
        paths[i].l = min(a, b) + 10000;
        paths[i].r = max(a, b) + 10000;
    }
    sort(paths, paths + n, byRight);
    int total = 0;
    for (int i = 0; i < n; ++i) {
        int l = paths[i].l, r = paths[i].r;
        int need = min(k, r - l + 1);
        for (int j = l; j <= r; ++j) if (chosen[j]) --need;
        for (int j = r; need > 0; --j) {
            if (!chosen[j]) { chosen[j] = true; --need; ++total; }
        }
    }
    printf("%d\n", total);
    for (int j = 0; j <= 20000; ++j)
        if (chosen[j]) printf("%d\n", j - 10000);
    return 0;
}
```

```cpp
// POJ 1752 - Advertisement
// Model: gpt-6-astra
// Independent control: longest-path difference constraints on prefix counts.
// Each path requires S[r+1]-S[l] >= min(K,r-l+1), and consecutive
// prefix counts differ by zero or one. Least feasible potentials minimize M.
#include <cstdio>
#include <algorithm>
#include <queue>
using namespace std;
const int V=20002;
struct Edge { int v,w,next; } edges[42010];
int head[V],dist[V],ec;
bool queued[V];
void add(int u,int v,int w) {
    edges[ec].v=v; edges[ec].w=w; edges[ec].next=head[u];head[u]=ec++;
}
int main() {
    int k,n;
    if(scanf("%d%d",&k,&n)!=2) return 0;
    fill(head,head+V,-1);
    for(int i=0;i<n;++i) {
        int a,b;scanf("%d%d",&a,&b);
        if(a>b) swap(a,b);
        add(a+10000,b+10001,min(k,b-a+1));
    }
    for(int i=0;i<V-1;++i) {add(i,i+1,0);add(i+1,i,-1);}
    queue<int> q;
    for(int i=0;i<V;++i) {q.push(i);queued[i]=true;}
    while(!q.empty()) {
        int u=q.front();q.pop();queued[u]=false;
        for(int e=head[u];e!=-1;e=edges[e].next) {
            int v=edges[e].v;
            if(dist[v]<dist[u]+edges[e].w) {
                dist[v]=dist[u]+edges[e].w;
                if(!queued[v]) {q.push(v);queued[v]=true;}
            }
        }
    }
    printf("%d\n",dist[V-1]);
    for(int i=0;i<V-1;++i) if(dist[i+1]>dist[i]) printf("%d\n",i-10000);
    return 0;
}
```
