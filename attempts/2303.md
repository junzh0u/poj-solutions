# 2303 — Russian Dolls

The dolls must be partitioned into two chains of exactly n dolls. Doll b fits inside a when b.h <= a.h - 2*a.w and b.d <= a.d - 2*a.w. Equality is allowed, confirmed by the original contest checker. Print each chain in descending height, separated by a hyphen, with blank lines between cases.

The independent solution constructs an edge between every pair that cannot nest. Those dolls must belong to different chains. A promised valid solution makes this graph bipartite. Each connected component has two possible orientations; subset-sum over its two color-class sizes picks exactly n dolls for the first set. Pairwise compatibility is sufficient because containment is transitive, and sorting by height gives the chain. Complexity is O(n^2) time and space.

// POJ 2303 - Russian Dolls
// Model: gpt-6-astra
// Incompatible dolls must have opposite colors. Each bipartite component
// can be flipped; subset-sum chooses flips giving exactly n dolls per set.
// Equality with the outer doll's hollow dimensions is allowed.
// Pairwise compatibility implies a nesting chain when sorted by height.
// O(n^2) time and space for 2n dolls.
#include <cstdio>
#include <cstring>
#include <vector>
#include <algorithm>
using namespace std;
struct Doll { int h,d,w; };
bool taller(const Doll &a,const Doll &b) { return a.h>b.h; }
bool fits(const Doll &a,const Doll &b) {
    return a.h<=b.h-2*b.w && a.d<=b.d-2*b.w;
}
int main() {
    int n; bool first=true;
    while (scanf("%d",&n)==1 && n) {
        vector<Doll> a(2*n);
        for(int i=0;i<2*n;++i) scanf("%d%d%d",&a[i].h,&a[i].d,&a[i].w);
        sort(a.begin(),a.end(),taller);
        vector<int> adj[200];
        for(int i=0;i<2*n;++i) for(int j=i+1;j<2*n;++j)
            if(!fits(a[j],a[i])) { adj[i].push_back(j); adj[j].push_back(i); }
        int color[200],comp[200],count[200][2],components=0;
        memset(color,-1,sizeof color); memset(count,0,sizeof count);
        for(int root=0;root<2*n;++root) if(color[root]<0) {
            vector<int> q(1,root); color[root]=0;
            for(unsigned k=0;k<q.size();++k) {
                int u=q[k]; comp[u]=components; ++count[components][color[u]];
                for(unsigned e=0;e<adj[u].size();++e) {
                    int v=adj[u][e];
                    if(color[v]<0) { color[v]=color[u]^1; q.push_back(v); }
                }
            }
            ++components;
        }
        int parent[201][101]; memset(parent,-1,sizeof parent); parent[0][0]=0;
        for(int c=0;c<components;++c) for(int s=0;s<=n;++s)
            if(parent[c][s]>=0) for(int side=0;side<2;++side) {
                int t=s+count[c][side];
                if(t<=n && parent[c+1][t]<0) parent[c+1][t]=side;
            }
        int chosen[200],s=n;
        for(int c=components-1;c>=0;--c) {
            chosen[c]=parent[c+1][s]; s-=count[c][chosen[c]];
        }
        if(!first) puts(""); first=false;
        for(int group=0;group<2;++group) {
            if(group) puts("-");
            for(int i=0;i<2*n;++i) if((color[i]!=chosen[comp[i]])==group)
                printf("%d %d %d\n",a[i].h,a[i].d,a[i].w);
        }
    }
    return 0;
}
