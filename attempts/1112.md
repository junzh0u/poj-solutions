# POJ 1112 — Team Them Up!

Parked 2026-08-06 as a **judge malfunction**, not a solve failure. Two submissions of an

## Statement

Split all N (≤ 100) people into exactly two non-empty teams so that inside each team every
member knows every other member, maximising the size of the smaller team. Print each team's
size followed by its members, or `No solution`.

Input gives, for each person i, the list of people i knows, terminated by `0`.

## Reading

"Knows" is **directed** — the discuss board records solvers losing submissions to the
assumption that it is symmetric. Two people can be teammates only if *both* directions hold,
since a team requires every member to know every other.

## Algorithm

Build the mutual-acquaintance graph M (edge i–j iff `know[i][j] && know[j][i]`). A team must
be a clique in M, i.e. an independent set in the complement G'. Two teams that are each
independent sets in G' are exactly a proper 2-colouring of G', so G' must be bipartite —
BFS each component, and an odd cycle means `No solution`.

Each component then contributes two colour classes of fixed sizes (a, b), and which class
joins team 1 is free per component. A knapsack DP over achievable team-1 sizes picks the
assignment minimising |2s − N|, excluding s = 0 and s = N so both teams stay non-empty.
Reconstruct the per-component choice by walking the DP table back.

O(N²) build, O(N·components) DP. Runs in ~3ms at N = 100.

```cpp
// POJ 1112 - Team Them Up!
// Model: claude-sonnet-5
// Approach: "Knows" is a directed relation (confirmed on the discuss board: several
// solvers WA'd assuming it was undirected). Two people can only be safe teammates
// if BOTH directions hold (i and j each list the other), since a team requires
// every member to know every other member. So build the "mutual acquaintance"
// graph M (i-j edge iff know[i][j] && know[j][i]); a team must be a clique in M,
// equivalently an independent set in the complement graph G'. Splitting everyone
// into two teams that are each independent sets in G' is exactly a proper
// 2-coloring of G' -> G' must be bipartite, checked per connected component via
// BFS (an odd cycle -> "No solution"). Each component has two color classes of
// fixed sizes (a,b); which class goes to team 1 is free per component, so we pick
// the assignment that makes the two teams as close in size as possible via a
// knapsack-style DP over achievable team-1 sizes (each component contributes
// either a or b), excluding sums of 0 or N since both teams need >=1 member.
#include <cstdio>
#include <vector>
#include <cmath>
using namespace std;

int main() {
    int n;
    if (scanf("%d", &n) != 1) return 0;
    static bool know[105][105];
    for (int i = 1; i <= n; ++i) {
        int x;
        while (scanf("%d", &x) == 1 && x != 0) {
            know[i][x] = true;
        }
    }

    static bool mutualEdge[105][105]; // complement graph G' edge (non-mutual acquaintance)
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= n; ++j) {
            if (i == j) continue;
            bool mutual = know[i][j] && know[j][i];
            mutualEdge[i][j] = !mutual;
        }
    }

    static int color[105];
    static int compId[105];
    for (int i = 1; i <= n; ++i) { color[i] = -1; compId[i] = -1; }

    int numComp = 0;
    vector<int> compA, compB; // sizes of color-0 and color-1 classes per component
    bool ok = true;

    static int queue_[105];
    for (int s = 1; s <= n && ok; ++s) {
        if (color[s] != -1) continue;
        int comp = numComp++;
        int qh = 0, qt = 0;
        queue_[qt++] = s;
        color[s] = 0;
        compId[s] = comp;
        int cntA = 0, cntB = 0;
        while (qh < qt) {
            int u = queue_[qh++];
            if (color[u] == 0) ++cntA; else ++cntB;
            for (int v = 1; v <= n; ++v) {
                if (v == u || !mutualEdge[u][v]) continue;
                if (color[v] == -1) {
                    color[v] = 1 - color[u];
                    compId[v] = comp;
                    queue_[qt++] = v;
                } else if (color[v] == color[u]) {
                    ok = false;
                }
            }
        }
        compA.push_back(cntA);
        compB.push_back(cntB);
    }

    if (!ok) {
        printf("No solution\n");
        return 0;
    }

    // DP: reachable[k][s] = true if after considering first k components, team-1
    // size s is achievable.
    vector<vector<bool> > reachable(numComp + 1, vector<bool>(n + 1, false));
    reachable[0][0] = true;
    for (int k = 1; k <= numComp; ++k) {
        int a = compA[k - 1], b = compB[k - 1];
        for (int s = 0; s <= n; ++s) {
            if (!reachable[k - 1][s]) continue;
            if (s + a <= n) reachable[k][s + a] = true;
            if (s + b <= n) reachable[k][s + b] = true;
        }
    }

    int best = -1;
    for (int s = 1; s <= n - 1; ++s) {
        if (!reachable[numComp][s]) continue;
        if (best == -1 || abs(2 * s - n) < abs(2 * best - n)) best = s;
    }

    if (best == -1) {
        printf("No solution\n");
        return 0;
    }

    // choice[k] = true means color-0 class of component k goes to team 1.
    vector<bool> choice(numComp, false);
    int s = best;
    for (int k = numComp; k >= 1; --k) {
        int a = compA[k - 1], b = compB[k - 1];
        if (s - a >= 0 && reachable[k - 1][s - a]) {
            choice[k - 1] = true;
            s -= a;
        } else {
            choice[k - 1] = false;
            s -= b;
        }
    }

    vector<int> team1, team2;
    for (int i = 1; i <= n; ++i) {
        int c = compId[i];
        bool colorZeroToTeam1 = choice[c];
        bool inTeam1 = (color[i] == 0) ? colorZeroToTeam1 : !colorZeroToTeam1;
        if (inTeam1) team1.push_back(i); else team2.push_back(i);
    }

    printf("%d", (int)team1.size());
    for (size_t i = 0; i < team1.size(); ++i) printf(" %d", team1[i]);
    printf("\n%d", (int)team2.size());
    for (size_t i = 0; i < team2.size(); ++i) printf(" %d", team2[i]);
    printf("\n");
    return 0;
}
```
