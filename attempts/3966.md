## Statement and algorithm

Input is exactly five distinct two-character cards. Output the ordinal of the hand among all 7462 poker ranking equivalence classes, weakest numbered 1. Categories ascend from high card, pair, two pairs, three of a kind, straight, flush, full house, four of a kind, to straight flush. Suits do not break ties. A2345 is the lowest straight (high card 5); no other wrapping is permitted.

Enumerate all nondecreasing sequences of five ranks in 2..14, excluding five equal ranks. Each gives a legal nonflush hand. If all ranks are distinct, also evaluate it as a flush. A key consists of the category followed by five base-15 tie-breaker digits. Multiplicity groups descend by frequency and then rank; flush and high-card ranks descend; straights use their high rank. Sorting and deduplicating the keys gives exactly 7462 entries. Evaluate the input and return its one-based lower_bound position. All arithmetic fits a signed 32-bit integer (keys are below 9*15^5 = 6834375).

The complete first implementation is preserved below, so a retry does not depend on any temporary file.

```cpp
// POJ 3966 - Hands of Poker
// Model: gpt-6.1-sol
// Enumerate legal rank multisets and their possible flush states, encode each
// category and its tie breakers lexicographically, then rank the unique keys.
// The wheel A2345 has high card 5; flushes require five distinct ranks.
#include <iostream>
#include <vector>
#include <algorithm>
#include <cstring>
using namespace std;
int score(const int *r, bool flush) {
    int count[15]={0}, tie[5]={0}, n=0, groups[5]={0};
    for(int i=0;i<5;i++) ++count[r[i]];
    for(int i=2;i<=14;i++) if(count[i]) ++groups[count[i]];
    int straight=0;
    if(groups[1]==5) {
        if(r[4]-r[0]==4) straight=r[4];
        else if(r[0]==2 && r[1]==3 && r[2]==4 && r[3]==5 && r[4]==14) straight=5;
    }
    int cat=0;
    if(straight && flush) {cat=8; tie[n++]=straight;}
    else if(groups[4]) cat=7;
    else if(groups[3] && groups[2]) cat=6;
    else if(flush) cat=5;
    else if(straight) {cat=4; tie[n++]=straight;}
    else if(groups[3]) cat=3;
    else if(groups[2]==2) cat=2;
    else if(groups[2]) cat=1;
    if(!n) {
        if(cat==5 || cat==0) for(int i=4;i>=0;i--) tie[n++]=r[i];
        else for(int c=4;c>=1;c--) for(int v=14;v>=2;v--) if(count[v]==c) tie[n++]=v;
    }
    int key=cat;
    for(int i=0;i<5;i++) key=key*15+tie[i];
    return key;
}
int main() {
    vector<int> keys;
    int r[5];
    for(r[0]=2;r[0]<=14;r[0]++)
    for(r[1]=r[0];r[1]<=14;r[1]++)
    for(r[2]=r[1];r[2]<=14;r[2]++)
    for(r[3]=r[2];r[3]<=14;r[3]++)
    for(r[4]=r[3];r[4]<=14;r[4]++) {
        if(r[0]==r[4]) continue;
        keys.push_back(score(r,false));
        bool distinct=true;
        for(int i=1;i<5;i++) if(r[i]==r[i-1]) distinct=false;
        if(distinct) keys.push_back(score(r,true));
    }
    sort(keys.begin(),keys.end());
    keys.erase(unique(keys.begin(),keys.end()),keys.end());
    string card, ranks="23456789TJQKA";
    char suit=0; bool flush=true;
    for(int i=0;i<5;i++) {
        if(!(cin>>card)) return 0;
        r[i]=int(ranks.find(card[0]))+2;
        if(i==0) suit=card[1];
        else if(card[1]!=suit) flush=false;
    }
    sort(r,r+5);
    cout << lower_bound(keys.begin(),keys.end(),score(r,flush))-keys.begin()+1 << '\n';
}
```
