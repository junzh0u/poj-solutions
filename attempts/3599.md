// POJ 3599 - Pumping Lemma
// Model: gpt-6-astra
// Independent control: compute shortest distances to acceptance by relaxation.
// Follow productive transitions until a state repeats, obtaining a loop.
// Append a shortest accepting suffix. The total length is at most 2n-1.
#include <cstdio>
#include <cstring>
#include <string>
#include <iostream>
using namespace std;
int tr[1001][26],d[1001],pos[1001];
int main() {
    int n,s,m,k,u,v; char c[10];
    if(scanf("%d%d",&n,&s)!=2) return 0;
    scanf("%d",&m);
    while(m--) { scanf("%d%s%d",&u,c,&v); tr[u][c[0]-'a']=v; }
    for(u=0;u<=n;++u) d[u]=100000;
    scanf("%d",&k); while(k--) { scanf("%d",&u); d[u]=0; }
    for(int it=0;it<n;++it) for(u=1;u<=n;++u) for(int j=0;j<26;++j)
        if(d[tr[u][j]]+1<d[u]) d[u]=d[tr[u][j]]+1;
    // Remove productive states with no surviving successor, leaving exactly
    // those from which an infinite productive walk is possible.
    int live[1001]; for(u=0;u<=n;++u) live[u]=(d[u]<100000);
    for(int it=0;it<n;++it) for(u=1;u<=n;++u) if(live[u]) {
        bool ok=false; for(int j=0;j<26;++j) if(live[tr[u][j]]) ok=true;
        if(!ok) live[u]=0;
    }
    if(!live[s]) { puts("*"); return 0; }
    memset(pos,-1,sizeof(pos)); string path; u=s;
    while(pos[u]<0) {
        pos[u]=path.size();
        for(int j=0;j<26;++j) if(live[tr[u][j]]) { path+=char('a'+j); u=tr[u][j]; break; }
    }
    string x=path.substr(0,pos[u]),y=path.substr(pos[u]),z;
    while(d[u]>0) for(int j=0;j<26;++j) if(d[tr[u][j]]+1==d[u]) { z+=char('a'+j); u=tr[u][j]; break; }
    cout<<x<<'('<<y<<')'<<z<<'\n';
}
