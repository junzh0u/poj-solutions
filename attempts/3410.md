// POJ 3410 - Split convex polygon
// Model: gpt-6.1-sol
// Match a shared cut chain in opposite boundary directions. Its internal
// angles complement to 2*pi, its endpoint angles sum to at most pi, and
// all vertices outside the cut must be convex. Normalize coordinates
// before comparing edge lengths; use atan2 for signed interior angles.
#include <cstdio>
#include <cmath>
#include <vector>
#include <algorithm>
using namespace std;
const double EPS=1e-9, TOL=1e-8;
struct P {
    double x,y;
    P(double a=0,double b=0):x(a),y(b){}
    P operator+(P b)const{return P(x+b.x,y+b.y);}
    P operator-(P b)const{return P(x-b.x,y-b.y);}
    P operator*(double t)const{return P(x*t,y*t);}
    bool operator<(P b)const{return x<b.x || (x==b.x && y<b.y);}
};
double cross(P a,P b){return a.x*b.y-a.y*b.x;}
double dot(P a,P b){return a.x*b.x+a.y*b.y;}
double len(P a){return sqrt(dot(a,a));}
bool solve(const vector<P>& a,const vector<P>& b){
    const double PI=acos(-1.0), E=1e-9;
    int n=a.size(),m=b.size();
    vector<double> aa(n),bb(m),al(n),bl(m);
    for(int i=0;i<n;++i){
        P u=a[i]-a[(i+n-1)%n],v=a[(i+1)%n]-a[i];
        aa[i]=PI-atan2(cross(u,v),dot(u,v));al[i]=len(v);
    }
    for(int i=0;i<m;++i){
        P u=b[i]-b[(i+m-1)%m],v=b[(i+1)%m]-b[i];
        bb[i]=PI-atan2(cross(u,v),dot(u,v));bl[i]=len(v);
    }
    for(int i=0;i<n;++i)for(int j=0;j<m;++j){
        if(aa[i]+bb[j]>PI+E)continue;
        for(int k=1;k<min(n,m);++k){
            int x=(i+k)%n,y=(j-k+m)%m;
            if(fabs(al[(x+n-1)%n]-bl[y])>E)break;
            if(aa[x]+bb[y]<=PI+E){
                bool ok=true;
                for(int t=k+1;t<n;++t)if(aa[(i+t)%n]>PI+E)ok=false;
                for(int t=1;t<m-k;++t)if(bb[(j+t)%m]>PI+E)ok=false;
                if(ok)return true;
            }
            if(fabs(aa[x]+bb[y]-2*PI)>E)break;
        }
    }
    return false;
}
int main(){
    int n,m;
    while(scanf("%d",&n)==1){
        vector<P>a(n);for(int i=0;i<n;++i)scanf("%lf%lf",&a[i].x,&a[i].y);
        if(scanf("%d",&m)!=1)return 0;
        vector<P>b(m);for(int i=0;i<m;++i)scanf("%lf%lf",&b[i].x,&b[i].y);
        double scale=0;
        P oa=a[0],ob=b[0];
        for(int i=0;i<n;++i){a[i]=a[i]-oa;scale=max(scale,len(a[i]));}
        for(int i=0;i<m;++i){b[i]=b[i]-ob;scale=max(scale,len(b[i]));}
        if(scale>0){
            for(int i=0;i<n;++i)a[i]=a[i]*(1.0/scale);
            for(int i=0;i<m;++i)b[i]=b[i]*(1.0/scale);
        }
        printf("%d\n",solve(a,b)?1:0);
    }
    return 0;
}
