# 1893 Monitoring Wheelchair Patients

The fresh gpt-6.1-sol attempt exhausted its five-submission cap on 2026-10-16 (POJ server dates), all under language 4, C++. Suggested park kind: model. No concrete failing case was recovered. A precision-sensitive output difference remains plausible, but has not been established as the cause.

The wheelchair starts at (200,0) in a closed [0,400] by [0,200] rectangle. Compass bearing zero points north, so a segment of length (end-start)*speed has displacement (length*sin(angle), length*cos(angle)). Idle gaps change elapsed time but not position. Total distance includes all motion after departure. While there has been no departure, the maximum Euclidean distance from the door occurs at a segment endpoint because squared distance is convex. When an endpoint is outside the convex rectangle, intersect that segment with each violated boundary and choose the earliest crossing. Crossing time includes the segment start time, not just cumulative moving time.

Final submitted source (25225229):

```cpp
// POJ 1893 - Monitoring Wheelchair Patients
// Model: gpt-6.1-sol
// Integrate each straight segment and clip the first departing segment
// against the four rectangle sides. The farthest distance on an inside
// segment occurs at an endpoint; total travel includes motion after exit.
// Intersect in elapsed seconds using velocities; use double and C++ as
// recommended by the problem discussion for its rounding-sensitive data.
#include <cstdio>
#include <cmath>
#include <algorithm>
using namespace std;
int main() {
    int n, cs=0;
    const double pi=acos(-1.0), eps=1e-8;
    while (scanf("%d", &n)==1 && n) {
        double x=200, y=0, total=0, farthest=0;
        double ex=0, ey=0, et=0;
        bool left=false;
        for (int i=0;i<n;++i) {
            double a,b,s; int bearing;
            scanf("%lf%lf%lf%d",&a,&b,&s,&bearing);
            double dist=(b-a)*s;
            double vx=s*sin(bearing*pi/180), vy=s*cos(bearing*pi/180);
            double dx=(b-a)*vx, dy=(b-a)*vy;
            double nx=x+dx, ny=y+dy;
            total+=dist;
            if (!left) {
                if (nx < 0 || nx > 400 || ny < 0 || ny > 200) {
                    double f=b-a;
                    if (nx < 0) f=min(f,-x/vx);
                    if (nx > 400) f=min(f,(400-x)/vx);
                    if (ny < 0) f=min(f,-y/vy);
                    if (ny > 200) f=min(f,(200-y)/vy);
                    ex=x+vx*f; ey=y+vy*f; et=a+f;
                    if (fabs(ex)<eps) ex=0;
                    if (fabs(ey)<eps) ey=0;
                    left=true;
                } else {
                    farthest=max(farthest,sqrt((nx-200)*(nx-200)+ny*ny));
                }
            }
            x=nx; y=ny;
        }
        printf("Case Number %d\n",++cs);
        if (left) printf("Left restricted area at point (%.1f,%.1f) and time %.1f sec.\n",ex,ey,et);
        else {
            printf("No departure from restricted area\n");
            printf("Maximum distance patient traveled from door was %.1f feet\n",farthest);
        }
        printf("Total distance traveled was %.1f feet\n",total);
        printf("***************************************\n");
    }
    return 0;
}
```

// POJ 1893 - Monitoring Wheelchair Patients
// Model: gpt-6-astra
// Clip the first departing segment against the closed rectangle. Squared
// distance is convex on each segment, so inside endpoints give its maximum.
// Parse recorded tenths exactly; accumulate total distance in hundredths.
// Use long double geometry and decimal half-up output to avoid binary ties.
#include <cstdio>
#include <cmath>
#include <algorithm>
using namespace std;
int tenth(const char *s) {
    int a=0;
    while(*s && *s!='.') a=a*10+(*s++-'0');
    return 10*a+(*s ? s[1]-'0':0);
}
double rounded(long double a) {
    return (double)(floorl(a*10+0.5L+1e-12L)/10);
}
int main() {
    int n,cs=0;
    const long double pi=acosl(-1.0L);
    while(scanf("%d",&n)==1 && n) {
        long double x=200,y=0,far=0,ex=0,ey=0,et=0;
        int total=0;
        bool departed=false;
        for(int i=0;i<n;++i) {
            char as[40],bs[40],ss[40];int d;
            scanf("%39s%39s%39s%d",as,bs,ss,&d);
            int a=tenth(as),b=tenth(bs),s=tenth(ss);
            total+=(b-a)*s;
            long double dx=(b-a)*s/100.0L*sinl(d*pi/180);
            long double dy=(b-a)*s/100.0L*cosl(d*pi/180);
            if(d%180==0) dx=0;
            if(d%180==90) dy=0;
            long double nx=x+dx,ny=y+dy;
            if(!departed && (nx < -1e-12L || nx>400+1e-12L || ny < -1e-12L || ny>200+1e-12L)) {
                long double f=1;
                if(nx<0) f=min(f,-x/dx);
                if(nx>400) f=min(f,(400-x)/dx);
                if(ny<0) f=min(f,-y/dy);
                if(ny>200) f=min(f,(200-y)/dy);
                ex=x+dx*f;ey=y+dy*f;et=(a+(b-a)*f)/10;
                departed=true;
            }
            x=nx;y=ny;
            far=max(far,sqrtl((x-200)*(x-200)+y*y));
        }
        printf("Case Number %d\n",++cs);
        if(departed) printf("Left restricted area at point (%.1f,%.1f) and time %.1f sec.\n",rounded(ex),rounded(ey),rounded(et));
        else {
            printf("No departure from restricted area\n");
            printf("Maximum distance patient traveled from door was %.1f feet\n",rounded(far));
        }
        printf("Total distance traveled was %d.%d feet\n",(total+5)/100,((total+5)/10)%10);
        printf("***************************************\n");
    }
    return 0;
}
