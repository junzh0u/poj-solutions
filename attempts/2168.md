# 2168 — Joke with Turtles

## Statement and algorithm

There are n turtles; equal coordinates form groups. A truthful claim (a,b) requires exactly a turtles ahead, b behind, and a group of size n-a-b. In rank order it occupies the half-open interval [a,n-b). Claims with a+b>=n cannot be truthful. Group identical intervals, with weight min(number of claimants, interval length), then find a maximum-weight collection of disjoint intervals. Unselected turtles fill the unused ranks and unused group capacity. Selected intervals reserve exactly their whole group sizes, so this completion always exists. Maximizing truthful turtles minimizes liars; any accidental truthful filler would contradict maximality.

The primary program uses weighted interval scheduling over right endpoints, stores predecessor positions and selected groups, and prints the complement of reconstructed truthful IDs. Every ID is one-based. The solution uses O(n log n) time and O(n) storage. No multiple-case framing is specified, so one case is read.

## Original submitted source

```cpp
// POJ 2168 - Joke with Turtles
// Model: gpt-6-astra
// A truthful (a,b) group occupies ranks [a,n-b), with capacity n-a-b.
// Weight each distinct interval by min(capacity, number of its claimants).
// Weighted interval scheduling maximizes truth-tellers; reconstruct their IDs.
// Equal coordinates are allowed, so a group may also contain lying fillers.
#include <iostream>
#include <vector>
#include <map>
#include <utility>
#include <algorithm>
using namespace std;
struct Group { int l, r; vector<int> ids; };
int main() {
    int n;
    if (!(cin >> n)) return 0;
    map<pair<int,int>, vector<int> > claims;
    for (int i=1; i<=n; ++i) {
        int a,b; cin >> a >> b;
        if (a+b<n) claims[make_pair(a,n-b)].push_back(i);
    }
    vector<Group> groups;
    vector<vector<int> > ending(n+1);
    for (map<pair<int,int>, vector<int> >::iterator it=claims.begin(); it!=claims.end(); ++it) {
        Group g; g.l=it->first.first; g.r=it->first.second; g.ids=it->second;
        ending[g.r].push_back((int)groups.size()); groups.push_back(g);
    }
    vector<int> dp(n+1,0), prev(n+1,0), chosen(n+1,-1);
    for (int r=1; r<=n; ++r) {
        dp[r]=dp[r-1]; prev[r]=r-1;
        for (int k=0; k<(int)ending[r].size(); ++k) {
            int id=ending[r][k]; const Group &g=groups[id];
            int weight=min(g.r-g.l,(int)g.ids.size());
            if (dp[g.l]+weight>dp[r]) {
                dp[r]=dp[g.l]+weight; prev[r]=g.l; chosen[r]=id;
            }
        }
    }
    vector<bool> truthful(n+1,false);
    for (int r=n; r>0; r=prev[r]) if (chosen[r]!=-1) {
        const Group &g=groups[chosen[r]];
        int weight=min(g.r-g.l,(int)g.ids.size());
        for (int j=0; j<weight; ++j) truthful[g.ids[j]]=true;
    }
    cout << n-dp[n];
    for (int i=1; i<=n; ++i) if (!truthful[i]) cout << ' ' << i;
    cout << '\n';
    return 0;
}
```

// POJ 2168 - Joke with Turtles
// Model: gpt-6-astra
// Control: C++98 translation of Roman Elizarov's NEERC 2004 jury solution.
// Source: https://neerc.ifmo.ru/archive/2004/problems/archive.rar
// Sort claims by a,b and compute the best suffix of rank positions.
#include <iostream>
#include <algorithm>
using namespace std;
struct Turtle { int i,a,b; bool operator<(const Turtle& o) const {
    return a!=o.a ? a<o.a : b<o.b;
}};
Turtle t[1002];
int z[1002],best[1002],j0s[1002],j1s[1002],nxt[1002];
bool truth[1002];
int main() {
    int n; if(!(cin>>n)) return 0;
    for(int i=0;i<n;++i) {t[i].i=i;cin>>t[i].a>>t[i].b;}
    t[n].i=n;t[n].a=n;t[n].b=0;
    sort(t,t+n+1);
    for(int i=0,a=0;i<=n;++i)
        while(a<=n && a<=t[i].a) z[a++]=i;
    for(int a=n-1;a>=0;--a) {
        best[a]=best[a+1];nxt[a]=a+1;
        for(int j=z[a];j<z[a+1];) {
            int j0=j,b=t[j].b,cnt=1;++j;
            while(j<=n && t[j].a==a && t[j].b==b) {++cnt;++j;}
            cnt=min(cnt,n-a-b);
            if(cnt>0 && cnt+best[n-b]>best[a]) {
                best[a]=cnt+best[n-b];j0s[a]=j0;j1s[a]=j0+cnt;nxt[a]=n-b;
            }
        }
    }
    for(int a=0;a<n;a=nxt[a])
        for(int j=j0s[a];j<j1s[a];++j) truth[t[j].i]=true;
    cout<<n-best[0];
    for(int i=0;i<n;++i) if(!truth[i]) cout<<' '<<i+1;
    cout<<'\n';
    return 0;
}
