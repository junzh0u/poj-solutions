## Statement and algorithm

Read one formula of at most 1000 characters. Variables are case-sensitive letters, ordered ASCII lexicographically in the output. Operator precedence is negation, conjunction, disjunction, implication, equivalence. Implication associates to the right; an equivalence chain means all adjacent operands are equal, not repeated binary equivalence. A valid formula prints `true`; otherwise print `false` and every occurring variable as a comma-separated assignment, with a colon before the first. A variable-free false formula prints only `false`.

The independent implementation parses these rules, interns simplified conjunction nodes, expresses disjunction/implication/equality through conjunction and negated literals, and produces Tseitin CNF. The formula root is required false. Complete conflict-driven SAT search uses unit propagation, first-UIP learned clauses, nonchronological backtracking, and variable activity. A satisfying model gives a counterexample; unsatisfiability establishes validity. It prints all occurring variables, including those eliminated by simplification.

#include <assert.h>
#include <stdio.h>
#include <ctype.h>
#include <string.h>
#include <stdlib.h>

#define MAXL 1000
#define MAXV (26*2+2)
#define MAXN (MAXL*10)
#define MAXQ (MAXN*2)
#define STKSIZE (1L << 24)

typedef enum tnode {
  Const, Var, And, Or, Not, Imply, Equiv
} tnode_t;

char s[MAXL+2], *p;

tnode_t op[MAXN];
int L[MAXN], R[MAXN];
char used[MAXN];
int n;

char sksp (void) {
  while (*p == ' ' || *p == '\t' || *p == '\r') p++;
  return *p;
}

void err (const char *str) {
  char *q;
  fputs (s, stdout);
  for (q = s; q < p; q++)
    putchar (*q < ' ' ? *q : ' ');
  printf ("^\n%s\n", str);
  fprintf (stderr, "In: %s%s\n", s, str);
  exit (1);
}

int newnode (tnode_t o, int x, int y) {
  int t;
  if (x > y && (o == And || o == Or || o == Equiv))
    { t=x; x=y; y=t; }
  for (t = 0; t < n; t++)
    if (op[t] == o && L[t] == x && R[t] == y) {
      used[t] = 1;  return t; }
  assert (n < MAXN);
  op[n] = o;  L[n] = x;  R[n] = y;  used[n] = 1;
  return n++;
}

int rexpr(void);
int rterm (void) {
  int c = sksp(), t;
  p++;
  if (isalpha(c)) return newnode (Var, c, 0);
  switch (c) {
  case '0': case '1':
    return c & 1;
  case '~':
    return newnode (Not, rterm(), 0);
  case '(':
    t = rexpr();
    if (*p != ')') err ("`)' expected"); else p++;
    return t;
  default:
    p--;
    err ("unknown character");
    return -1;
  }
}

int r_and (void) {
  int r = rterm();
  while (sksp() == '&') {
    p++;  r = newnode (And, r, rterm()); }
  return r;
}

int r_or (void) {
  int r = r_and();
  while (sksp() == '|') {
    p++;  r = newnode (Or, r, r_and()); }
  return r;
}

int r_impl (void) {
  int r = r_or();
  if (sksp() == '=' && p[1] == '>') {
    p += 2;  r = newnode (Imply, r, r_impl()); }
  return r;
}

int rexpr (void) {
  int x = -1, z, r = r_impl();
  while (sksp() == '=' && p[1] != '>') {
    p++;  z = r_impl();
    r = (x < 0) ? newnode (Equiv, r, z) :
      newnode (And, r, newnode (Equiv, x, z));
    x = z;
  }
  return r;
}

    Q2[q2w++] = x;  assert (q2w <= MAXQ); }
  else {
    Q[qw++] = x;  assert (qw <= MAXQ); }
}
