// POJ 1735 - A Game on the Chessboard
// Model: gpt-6-astra
// Breadth-first search of the 16-bit board masks gives a shortest sequence.
// Store the preceding mask and exchanged edge to reconstruct legal moves.
// Equal-colour exchanges cannot improve a shortest path and are omitted.
#include <cstdio>
#include <cstring>
int pre[65536], how[65536], que[65536], path[65536];
int a[24], b[24];
int readboard() {
    int mask=0; char row[20];
    for(int r=0;r<4;++r) {
        if(scanf("%19s",row)!=1) return -1;
        for(int c=0;c<4;++c) if(row[c]=='1') mask|=1<<(r*4+c);
    }
    return mask;
}
int main() {
    int start=readboard(), goal=readboard();
    if(start<0 || goal<0) return 0;
    int edges=0;
    for(int r=0;r<4;++r) for(int c=0;c<4;++c) {
        int u=r*4+c;
        if(c<3) { a[edges]=u; b[edges++]=u+1; }
        if(r<3) { a[edges]=u; b[edges++]=u+4; }
    }
    memset(pre,-1,sizeof(pre));
    int head=0,tail=0; que[tail++]=start; pre[start]=start;
    while(head<tail && pre[goal]<0) {
        int u=que[head++];
        for(int e=0;e<edges;++e) {
            if(((u>>a[e])&1)==((u>>b[e])&1)) continue;
            int v=u^(1<<a[e])^(1<<b[e]);
            if(pre[v]>=0) continue;
            pre[v]=u; how[v]=e; que[tail++]=v;
        }
    }
    int n=0;
    for(int u=goal;u!=start;u=pre[u]) path[n++]=how[u];
    printf("%d\n",n);
    for(int i=n-1;i>=0;--i) {
        int e=path[i];
        printf("%d %d %d %d\n",a[e]/4+1,a[e]%4+1,b[e]/4+1,b[e]%4+1);
    }
    return 0;
}
