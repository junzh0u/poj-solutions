```cpp
// POJ 2169 - Kingdom of Magic
// Model: gpt-6-astra
// Dijkstra on ordered neighboring city pairs. Each traveler either stays
// or uses an edge; a simultaneous swap through the same portal is forbidden.
// Transition cost counts individual portal uses, not simultaneous rounds.
#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
bool edge[101][101];
int main() {
    int n,m,a,b,c,d;
    if(scanf("%d%d%d%d%d%d",&n,&m,&a,&b,&c,&d)!=6) return 0;
    for(int i=0;i<m;++i) { int x,y; scanf("%d%d",&x,&y); edge[x][y]=edge[y][x]=true; }
    vector<pair<int,int> > s;
    for(int x=1;x<=n;++x) for(int y=1;y<=n;++y)
        if(x!=y && edge[x][y]) s.push_back(make_pair(x,y));
    int count=s.size(), start=-1, goal=-1;
    for(int i=0;i<count;++i) {
        if(s[i]==make_pair(a,b)) start=i;
        if(s[i]==make_pair(c,d)) goal=i;
    }
    if(start<0 || goal<0) return 0;
    vector<int> dist(count,1000000000),prev(count,-1),used(count,0);
    dist[start]=0;
    for(int step=0;step<count;++step) {
        int u=-1;
        for(int i=0;i<count;++i) if(!used[i] && (u<0 || dist[i]<dist[u])) u=i;
        if(u<0 || dist[u]==1000000000 || u==goal) break;
        used[u]=1;
        int x=s[u].first,y=s[u].second;
        for(int v=0;v<count;++v) if(!used[v]) {
            int p=s[v].first,q=s[v].second;
            if(p==y && q==x) continue;
            if((p==x || edge[x][p]) && (q==y || edge[y][q])) {
                int nd=dist[u]+(p!=x)+(q!=y);
                if(nd<dist[v]) { dist[v]=nd; prev[v]=u; }
            }
        }
    }
    vector<int> path;
    for(int u=goal;u!=-1;u=prev[u]) path.push_back(u);
    reverse(path.begin(),path.end());
    printf("%d %d\n",dist[goal],(int)path.size());
    for(unsigned i=0;i<path.size();++i) printf("%d %d\n",s[path[i]].first,s[path[i]].second);
    return 0;
}
```
